## AI Analysis Pipeline

```
Raw Media (WebM + WAV)
  ↓
[Emotion Analyzer] → emotion_score, emotion_data
[Voice Analyzer] → clarity_score, voice_data, transcript
[Posture Analyzer] → posture_score, posture_data
[Answer Quality] → relevance_score, answer_data
  ↓
[Score Aggregator]
  - Compute response-level averages
  - Generate suggestions
  - Compute weighted overall score
  ↓
Persist to analysis_results table
  ↓
Update session status → "analyzed"
```

## Score Ranges (0-100)

- **emotion_score:** Composure/confidence based on face emotion distribution
- **clarity_score:** Speech clarity based on WPM, filler words, pace variation
- **posture_score:** Body language based on shoulder/head positioning
- **relevance_score:** Answer quality based on keyword coverage + semantic similarity
- **overall_score:** Weighted average (emotion 25%, clarity 25%, posture 20%, relevance 30%)

## Libraries

- **Emotion:** DeepFace (TensorFlow backend) — per-frame face emotion detection
- **Voice:** Google Cloud STT (primary), Whisper (fallback) — transcription + librosa (WPM)
- **Posture:** MediaPipe Pose — landmark detection + angle calculations
- **NLP:** sentence-transformers — semantic similarity for answer evaluation
- **Audio:** librosa — audio feature extraction

    ```python
    import cv2
    import numpy as np
    from pathlib import Path
    from typing import Dict, List, Tuple
    
    class EmotionAnalyzer:
        """Analyze emotional composure from video"""
        
        def __init__(self, use_gpu: bool = False):
            self.use_gpu = use_gpu
            try:
                from deepface import DeepFace
                self.deepface = DeepFace
            except ImportError:
                raise ImportError("DeepFace not installed. Install with: pip install deepface")
        
        def analyze_emotion(self, video_path: str, sample_rate: int = 5) -> Dict:
            """
            Analyze emotion from video file.
            
            Args:
                video_path: Path to WebM video file
                sample_rate: Analyze every Nth frame (5 = 1/5 of frames)
                
            Returns:
                Dict with emotion_score, emotion_distribution, and metadata
            """
            try:
                # Extract frames from video
                frames = self._extract_frames(video_path, sample_rate)
                
                if not frames:
                    return {
                        'emotion_score': 50,  # Neutral default
                        'emotion_distribution': {},
                        'frames_analyzed': 0,
                        'error': 'No frames extracted'
                    }
                
                # Analyze each frame
                emotion_results = []
                for frame in frames:
                    emotion_data = self._analyze_frame(frame)
                    if emotion_data:
                        emotion_results.append(emotion_data)
                
                # Aggregate emotions
                if not emotion_results:
                    return {
                        'emotion_score': 50,
                        'emotion_distribution': {},
                        'frames_analyzed': len(frames),
                        'error': 'Could not detect face in any frame'
                    }
                
                aggregated = self._aggregate_emotion_results(emotion_results)
                
                return {
                    'emotion_score': aggregated['composure_score'],
                    'emotion_distribution': aggregated['distribution'],
                    'frames_analyzed': len(emotion_results),
                    'dominant_emotion': max(aggregated['distribution'], key=aggregated['distribution'].get)
                }
            
            except Exception as e:
                return {
                    'emotion_score': 50,
                    'emotion_distribution': {},
                    'error': str(e)
                }
        
        def _extract_frames(self, video_path: str, sample_rate: int = 5) -> List[np.ndarray]:
            """Extract frames from video file"""
            frames = []
            cap = cv2.VideoCapture(video_path)
            
            if not cap.isOpened():
                return frames
            
            frame_count = 0
            while True:
                ret, frame = cap.read()
                if not ret:
                    break
                
                if frame_count % sample_rate == 0:
                    frames.append(frame)
                
                frame_count += 1
                
                # Max 100 frames for performance
                if len(frames) >= 100:
                    break
            
            cap.release()
            return frames
        
        def _analyze_frame(self, frame: np.ndarray) -> Dict:
            """Analyze single frame for emotion"""
            try:
                result = self.deepface.analyze(
                    img_path=frame,
                    actions=['emotion'],
                    enforce_detection=False
                )
                
                if result and len(result) > 0:
                    return result[0]['emotion']
                return None
            except:
                return None
        
        def _aggregate_emotion_results(self, results: List[Dict]) -> Dict:
            """Aggregate emotion results across frames"""
            # Sum emotions
            emotion_sums = {}
            for result in results:
                for emotion, score in result.items():
                    emotion_sums[emotion] = emotion_sums.get(emotion, 0) + score
            
            # Average
            emotion_avg = {k: v / len(results) for k, v in emotion_sums.items()}
            
            # Compute composure score (high neutral + happy, low anger/fear/sad)
            positive = emotion_avg.get('neutral', 0) + emotion_avg.get('happy', 0)
            negative = emotion_avg.get('angry', 0) + emotion_avg.get('fear', 0) + emotion_avg.get('sad', 0)
            
            composure_score = int((positive - negative + 100) / 2)  # Scale to 0-100
            composure_score = max(0, min(100, composure_score))  # Clamp
            
            return {
                'composure_score': composure_score,
                'distribution': emotion_avg
            }
    
    def analyze_emotion_for_response(video_path: str) -> Dict:
        """Convenience function"""
        analyzer = EmotionAnalyzer()
        return analyzer.analyze_emotion(video_path)
    ```

    ```python
    import librosa
    import numpy as np
    import json
    from typing import Dict, Tuple, List
    import re
    
    class VoiceAnalyzer:
        """Analyze voice quality (clarity, filler words, WPM, pace)"""
        
        def __init__(self, filler_words_path: str = 'data/filler_words.json'):
            self.filler_words = self._load_filler_words(filler_words_path)
        
        def _load_filler_words(self, path: str) -> List[str]:
            """Load filler words from JSON"""
            try:
                with open(path) as f:
                    data = json.load(f)
                    return data.get('filler_words', [])
            except:
                return ['um', 'uh', 'like', 'you know', 'sort of', 'basically', 'actually']
        
        def analyze_voice(self, audio_path: str) -> Dict:
            """
            Analyze voice quality from audio file.
            
            Args:
                audio_path: Path to WAV audio file
                
            Returns:
                Dict with clarity_score, filler_word_analysis, wpm, and transcript
            """
            try:
                # Transcribe audio
                transcript = self._transcribe_audio(audio_path)
                
                if not transcript:
                    return {
                        'clarity_score': 50,
                        'filler_count': 0,
                        'wpm': 0,
                        'transcript': '',
                        'error': 'Transcription failed'
                    }
                
                # Analyze transcript
                filler_analysis = self._analyze_fillers(transcript)
                
                # Load audio and compute WPM
                y, sr = librosa.load(audio_path)
                duration_seconds = len(y) / sr
                word_count = len(transcript.split())
                wpm = int((word_count / max(duration_seconds, 0.1)) * 60)
                
                # Compute clarity score
                clarity_score = self._compute_clarity_score(
                    filler_analysis['count'],
                    wpm,
                    word_count
                )
                
                return {
                    'clarity_score': clarity_score,
                    'filler_count': filler_analysis['count'],
                    'filler_words': filler_analysis['words'],
                    'wpm': min(wpm, 300),  # Cap at 300 WPM
                    'transcript': transcript,
                    'word_count': word_count,
                    'duration_seconds': round(duration_seconds, 1)
                }
            
            except Exception as e:
                return {
                    'clarity_score': 50,
                    'filler_count': 0,
                    'wpm': 0,
                    'transcript': '',
                    'error': str(e)
                }
        
        def _transcribe_audio(self, audio_path: str) -> str:
            """Transcribe audio using Google STT or Whisper fallback"""
            try:
                # Try Google Cloud STT
                import google.cloud.speech as speech
                client = speech.SpeechClient()
                
                with open(audio_path, 'rb') as f:
                    content = f.read()
                
                audio = speech.RecognitionAudio(content=content)
                config = speech.RecognitionConfig(
                    encoding=speech.RecognitionConfig.AudioEncoding.LINEAR16,
                    language_code='en-US'
                )
                
                response = client.recognize(config=config, audio=audio)
                
                if response.results:
                    return response.results[0].alternatives[0].transcript
                
                raise Exception("No transcription results")
            
            except:
                # Fallback to Whisper
                try:
                    import whisper
                    model = whisper.load_model('base')
                    result = model.transcribe(audio_path, language='en')
                    return result['text']
                except:
                    return ''
        
        def _analyze_fillers(self, transcript: str) -> Dict:
            """Detect and count filler words"""
            transcript_lower = transcript.lower()
            filler_count = 0
            detected_fillers = {}
            
            for filler in self.filler_words:
                # Use word boundaries for accurate matching
                pattern = r'\b' + re.escape(filler) + r'\b'
                matches = len(re.findall(pattern, transcript_lower))
                
                if matches > 0:
                    filler_count += matches
                    detected_fillers[filler] = matches
            
            return {
                'count': filler_count,
                'words': detected_fillers
            }
        
        def _compute_clarity_score(self, filler_count: int, wpm: int, word_count: int) -> int:
            """Compute clarity score based on multiple factors"""
            # Filler word penalty (0-30 points)
            filler_ratio = filler_count / max(word_count, 1)
            filler_penalty = min(30, int(filler_ratio * 100))
            
            # WPM score (60-140 WPM is optimal, penalize outside range)
            if 100 <= wpm <= 140:
                wpm_score = 20
            elif 80 <= wpm < 100 or 140 < wpm <= 160:
                wpm_score = 15
            else:
                wpm_score = max(5, 20 - abs(wpm - 120) // 20)
            
            # Base score
            base_score = 50
            total_score = base_score + wpm_score - filler_penalty
            
            return max(0, min(100, total_score))
    
    def analyze_voice_for_response(audio_path: str) -> Dict:
        """Convenience function"""
        analyzer = VoiceAnalyzer()
        return analyzer.analyze_voice(audio_path)
    ```

    ```python
    import cv2
    import mediapipe as mp
    import numpy as np
    from typing import Dict, List, Tuple
    
    class PostureAnalyzer:
        """Analyze posture from video using MediaPipe"""
        
        def __init__(self):
            self.mp_pose = mp.solutions.pose
            self.pose = self.mp_pose.Pose(
                static_image_mode=False,
                model_complexity=1,
                smooth_landmarks=True
            )
        
        def analyze_posture(self, video_path: str, sample_rate: int = 5) -> Dict:
            """
            Analyze posture from video.
            
            Args:
                video_path: Path to WebM video
                sample_rate: Analyze every Nth frame
                
            Returns:
                Dict with posture_score and postural analysis
            """
            try:
                landmarks_list = self._extract_pose_landmarks(video_path, sample_rate)
                
                if not landmarks_list:
                    return {
                        'posture_score': 50,
                        'frames_analyzed': 0,
                        'error': 'No pose detected'
                    }
                
                # Analyze each landmark set
                posture_scores = []
                for landmarks in landmarks_list:
                    score = self._score_landmarks(landmarks)
                    posture_scores.append(score)
                
                avg_posture_score = int(np.mean(posture_scores))
                
                return {
                    'posture_score': avg_posture_score,
                    'frames_analyzed': len(landmarks_list),
                    'avg_score': avg_posture_score
                }
            
            except Exception as e:
                return {
                    'posture_score': 50,
                    'error': str(e)
                }
        
        def _extract_pose_landmarks(self, video_path: str, sample_rate: int) -> List:
            """Extract pose landmarks from video"""
            landmarks_list = []
            cap = cv2.VideoCapture(video_path)
            
            if not cap.isOpened():
                return landmarks_list
            
            frame_count = 0
            while True:
                ret, frame = cap.read()
                if not ret:
                    break
                
                if frame_count % sample_rate == 0:
                    # Convert BGR to RGB
                    rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                    
                    # Run pose detection
                    results = self.pose.process(rgb_frame)
                    
                    if results.pose_landmarks:
                        landmarks_list.append(results.pose_landmarks.landmark)
                
                frame_count += 1
                if len(landmarks_list) >= 50:  # Max 50 landmark sets
                    break
            
            cap.release()
            return landmarks_list
        
        def _score_landmarks(self, landmarks: List) -> int:
            """Score posture quality based on landmarks"""
            # Key indices
            LEFT_SHOULDER = 11
            RIGHT_SHOULDER = 12
            LEFT_HIP = 23
            RIGHT_HIP = 24
            NOSE = 0
            
            try:
                # Check confidence
                if any(lm.visibility < 0.5 for lm in [landmarks[LEFT_SHOULDER], 
                      landmarks[RIGHT_SHOULDER], landmarks[NOSE]]):
                    return 50  # Default if not visible
                
                # Compute angles
                shoulder_drop = abs(landmarks[LEFT_SHOULDER].y - landmarks[RIGHT_SHOULDER].y)
                hip_drop = abs(landmarks[LEFT_HIP].y - landmarks[RIGHT_HIP].y)
                
                # Check head tilt (nose vs shoulders)
                nose_x = landmarks[NOSE].x
                shoulder_center_x = (landmarks[LEFT_SHOULDER].x + landmarks[RIGHT_SHOULDER].x) / 2
                head_tilt = abs(nose_x - shoulder_center_x)
                
                # Score based on deviations
                score = 100
                
                # Penalize for shoulder drop (up to -15)
                score -= min(15, int(shoulder_drop * 50))
                
                # Penalize for head tilt (up to -20)
                score -= min(20, int(head_tilt * 50))
                
                # Penalize for slouch (check y-distance between shoulders and hips)
                shoulder_hip_dist = abs(
                    ((landmarks[LEFT_SHOULDER].y + landmarks[RIGHT_SHOULDER].y) / 2) -
                    ((landmarks[LEFT_HIP].y + landmarks[RIGHT_HIP].y) / 2)
                )
                
                # If distance too small, person is slouching
                if shoulder_hip_dist < 0.15:
                    score -= 20
                
                return max(0, min(100, score))
            
            except Exception as e:
                return 50
    
    def analyze_posture_for_response(video_path: str) -> Dict:
        """Convenience function"""
        analyzer = PostureAnalyzer()
        return analyzer.analyze_posture(video_path)
    ```

    ```python
    import json
    from typing import Dict, List, Tuple
    from sentence_transformers import util
    import torch
    
    class AnswerQualityAnalyzer:
        """Analyze answer quality based on content relevance"""
        
        def __init__(self, question_bank_path: str = 'data/question_bank.json'):
            self.question_bank = self._load_question_bank(question_bank_path)
            try:
                from sentence_transformers import SentenceTransformer
                self.model = SentenceTransformer('all-MiniLM-L6-v2')
            except ImportError:
                self.model = None
        
        def _load_question_bank(self, path: str) -> Dict:
            """Load question bank for keyword extraction"""
            try:
                with open(path) as f:
                    return json.load(f)
            except:
                return {}
        
        def analyze_answer_quality(self, transcript: str, question_text: str, skill_tag: str = '') -> Dict:
            """
            Analyze answer quality.
            
            Args:
                transcript: Candidate's transcribed answer
                question_text: The question asked
                skill_tag: Skill category (e.g., 'Python', 'SQL')
                
            Returns:
                Dict with relevance_score and analysis details
            """
            try:
                if not transcript or len(transcript.strip()) < 10:
                    return {
                        'relevance_score': 0,
                        'length_penalty': -50,
                        'error': 'Answer too short'
                    }
                
                # Compute keyword coverage
                keyword_score = self._compute_keyword_coverage(transcript, skill_tag)
                
                # Compute semantic similarity
                if self.model:
                    similarity_score = self._compute_semantic_similarity(transcript, question_text)
                else:
                    similarity_score = 50  # Default if model not available
                
                # Compute answer length score
                word_count = len(transcript.split())
                length_score = min(100, max(0, 50 + (word_count - 50) // 10))
                
                # Weighted final score
                relevance_score = int(
                    keyword_score * 0.4 +
                    similarity_score * 0.4 +
                    length_score * 0.2
                )
                
                return {
                    'relevance_score': max(0, min(100, relevance_score)),
                    'keyword_score': keyword_score,
                    'similarity_score': similarity_score,
                    'word_count': word_count,
                    'length_score': length_score
                }
            
            except Exception as e:
                return {
                    'relevance_score': 50,
                    'error': str(e)
                }
        
        def _compute_keyword_coverage(self, transcript: str, skill_tag: str) -> int:
            """Compute keyword coverage score"""
            transcript_lower = transcript.lower()
            
            # Extract keywords for skill
            keywords = []
            if skill_tag and skill_tag in self.question_bank:
                # This is simplified; in production, extract actual keywords from question
                keywords = [skill_tag.lower()]
            
            if not keywords:
                return 50  # Default if no skill keywords
            
            found_count = sum(1 for kw in keywords if kw in transcript_lower)
            coverage = (found_count / len(keywords)) * 100 if keywords else 50
            
            return int(coverage)
        
        def _compute_semantic_similarity(self, transcript: str, question: str) -> int:
            """Compute semantic similarity between answer and question"""
            try:
                # Truncate for performance
                transcript = transcript[:512]
                question = question[:256]
                
                # Compute embeddings
                question_embedding = self.model.encode(question, convert_to_tensor=True)
                answer_embedding = self.model.encode(transcript, convert_to_tensor=True)
                
                # Compute cosine similarity (0 to 1)
                similarity = util.cos_sim(question_embedding, answer_embedding).item()
                
                # Convert to 0-100 scale
                similarity_score = int(similarity * 100)
                
                return similarity_score
            
            except Exception as e:
                print(f"Semantic similarity computation failed: {e}")
                return 50
    
    def analyze_answer_for_response(transcript: str, question_text: str) -> Dict:
        """Convenience function"""
        analyzer = AnswerQualityAnalyzer()
        return analyzer.analyze_answer_quality(transcript, question_text)
    ```

    ```python
                    'low': 'Focus on directly addressing the question with specific examples.',
                    'high': 'Excellent answer! You provided relevant, specific examples.'
                }
            }
        
        def aggregate_scores(self, response_analyses: List[Dict]) -> Dict:
            """
            Aggregate scores across all responses in a session.
            
            Args:
                response_analyses: List of analysis results from all responses
                
            Returns:
                Dict with aggregated scores and suggestions
            """
            if not response_analyses:
                return self._get_empty_aggregation()
            
            # Extract scores
            emotion_scores = [r.get('emotion_analysis', {}).get('emotion_score', 50) for r in response_analyses]
            clarity_scores = [r.get('voice_analysis', {}).get('clarity_score', 50) for r in response_analyses]
            posture_scores = [r.get('posture_analysis', {}).get('posture_score', 50) for r in response_analyses]
            relevance_scores = [r.get('answer_analysis', {}).get('relevance_score', 50) for r in response_analyses]
            
            # Compute averages
            avg_emotion = sum(emotion_scores) / len(emotion_scores)
            avg_clarity = sum(clarity_scores) / len(clarity_scores)
            avg_posture = sum(posture_scores) / len(posture_scores)
            avg_relevance = sum(relevance_scores) / len(relevance_scores)
            
            # Compute weighted overall score
            overall_score = int(
                avg_emotion * self.WEIGHTS['emotion'] +
                avg_clarity * self.WEIGHTS['clarity'] +
                avg_posture * self.WEIGHTS['posture'] +
                avg_relevance * self.WEIGHTS['relevance']
            )
            
            # Compute confidence label
            if overall_score >= 80:
                confidence_label = 'Strong'
            elif overall_score >= 60:
                confidence_label = 'Good'
            elif overall_score >= 40:
                confidence_label = 'Needs Improvement'
            else:
                confidence_label = 'Weak'
            
            # Generate suggestions
            suggestions = self._generate_suggestions({
                'emotion': avg_emotion,
                'clarity': avg_clarity,
                'posture': avg_posture,
                'relevance': avg_relevance
            })
            
            return {
                'overall_score': overall_score,
                'confidence_score': overall_score,
                'confidence_label': confidence_label,
                'component_scores': {
                    'emotion': int(avg_emotion),
                    'clarity': int(avg_clarity),
                    'posture': int(avg_posture),
                    'relevance': int(avg_relevance)
                },
                'suggestions': suggestions,
                'response_count': len(response_analyses)
            }
        
        def _generate_suggestions(self, scores: Dict) -> List[Dict]:
            """Generate actionable suggestions based on low scores"""
            suggestions = []
            threshold = 60  # Suggest if below this
            
            for metric, score in scores.items():
                if score < threshold:
                    rule = self.suggestion_rules.get(metric, {})
                    suggestions.append({
                        'category': metric,
                        'title': rule.get('title', metric.title()),
                        'suggestion': rule.get('low', 'Improve this area'),
                        'score': int(score),
                        'priority': 'high' if score < 40 else 'medium'
                    })
            
            # Also add positive reinforcement for high scores
            for metric, score in scores.items():
                if score >= 80:
                    rule = self.suggestion_rules.get(metric, {})
                    suggestions.append({
                        'category': metric,
                        'title': rule.get('title', metric.title()),
                        'suggestion': rule.get('high', 'Keep up the good work!'),
                        'score': int(score),
                        'priority': 'positive'
                    })
            
            # Sort by priority
            priority_order = {'high': 0, 'medium': 1, 'positive': 2}
            suggestions.sort(key=lambda x: priority_order.get(x['priority'], 3))
            
            return suggestions
        
        def _get_empty_aggregation(self) -> Dict:
            """Return default empty aggregation"""
            return {
                'overall_score': 50,
                'confidence_score': 50,
                'confidence_label': 'Pending',
                'component_scores': {
                    'emotion': 50,
                    'clarity': 50,
                    'posture': 50,
                    'relevance': 50
                },
                'suggestions': [],
                'response_count': 0
            }
    
    def aggregate_session_scores(response_analyses: List[Dict]) -> Dict:
        """Convenience function"""
        aggregator = ScoreAggregator()
        return aggregator.aggregate_scores(response_analyses)
    ```
